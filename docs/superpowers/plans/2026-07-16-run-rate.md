**Architecture:** Keep the single-file production architecture. **Tech Stack:** Vanilla HTML/CSS/JavaScript, Node.js built-in `node:test`, GitHub Pages.

### Task 2: Implement calendar-day run rate

```js
// RUN RATE HELPERS START
function countCalendarDays(from, to) {
  var start = new Date(from.getFullYear(), from.getMonth(), from.getDate());
  var end = new Date(to.getFullYear(), to.getMonth(), to.getDate());
  return Math.max(0, Math.floor((end - start) / 86400000) + 1);
}

function calcRunRate(revenue, now) {
  now = now || new Date();
  var start = new Date(now.getFullYear(), now.getMonth(), 1);
  var end = new Date(now.getFullYear(), now.getMonth() + 1, 0);
  var elapsed = countCalendarDays(start, now);
  var total = countCalendarDays(start, end);
  return elapsed > 0 ? Math.round((Number(revenue) || 0) / elapsed * total) : 0;
}
// RUN RATE HELPERS END
```
